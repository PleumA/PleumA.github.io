## Decisions (where the spec left a choice)

1. **Names (M-names):** strip `: ; , \r \n` (each run becomes one space), collapse whitespace, trim. Empty result is ignored. Duplicates are exact-match and case-sensitive; `addDoctor` toasts, import drops them silently. The same `sanitizeName` cleans a new role name typed into the role prompt. (Strip, not reject: a pasted `Dr. A; Dr. B` stays usable.)
2. **Import (C5):** a wrong top-level or container type (`doctors`, `offData`, `extraSlotsData`, `manualOverrides`, `inputs`, `checkboxes`) rejects the whole file with a toast and changes nothing. Bad entries inside valid containers are dropped or coerced. `manualOverrides` day and slot keys must be digits (this also blocks `__proto__`). If anything throws after state was assigned, the five data globals roll back.
3. **Special doctor (C4):** one shared `strictOk(doc, slotIdx)` closure holds the cascade's level-1 hard rules. It gates both the pre-pick and the moment the special doctor is placed in a slot. If the special doctor is not eligible, the slot falls through to the normal cascade; the locked doctor may still be picked there like anyone else.
4. **Inline handler arguments:** structured-input handlers keep their `(name, delta)` signatures, because `tests/structuredInputs.test.js` calls them by name. The name is passed as `esc(JSON.stringify(name))` (helper `jsArg`) instead of `'${name}'`. This is as safe as passing an index and breaks nothing. Doctor-row handlers that already pass indexes are untouched.
5. **Helpers:** `sanitizeName` and `parseISODate` live in `translations.js` (the existing shared-globals file, where `esc` is) and are added to `.eslintrc.cjs` globals.
6. **Period key (I1):** `"<month>-<AD year>"`, or `"<start>_<end>"` in custom mode. A period change clears overrides and the undo stack and shows an info toast, but only if overrides existed. An import sets the key to `null`, meaning "adopt the next run's period", so imported overrides apply.
7. **Regenerate queue (I2):** one boolean `regenPending` and a `do/while` around the existing body (renamed `runScheduleOnce`). All inputs, including `parseUIConfig()`, are read before the first `await`.
8. **Year (M-year):** valid means `calcYear >= 1000` after the BE conversion. Checked at the start of a standard-mode run, before the period check.
9. **Service worker (I9):** same-origin GET is network-first with cache fallback (not stale-while-revalidate: a deploy must show on the next load, and an offline start still works). CDN files are version-pinned, so cache-first. `CACHE_NAME` becomes `doctor-scheduler-v4`. `lucide` is pinned to **1.55.0**: that is what `@latest` resolves to today (npm `latest` dist-tag, 2026-10-10). All 24 `data-lucide` names used by `index.html` and `app.js` were checked to render with it.
10. **Off ranges (I8):** a numeric range is clamped to `numDays`. A custom range longer than 366 days is skipped after the existing ">90 days" warning. Range expansion steps by calendar day (see Review Focus 3).
11. **Calendar pop-up:** drop the `'noopener'` feature string and set `win.opener = null` instead. The `noopener` feature makes `window.open` always return `null`, which would make blocked-popup detection impossible.
12. **Aria labels:** reuse existing keys through `data-i18n-aria` (`darkModeTitle`, `langSwitchTitle`, `manualBtnTitle`, `calcBtn`), so no new strings.
13. **Run-tests strictness:** after the noise is fixed, `run-tests.js` treats `TypeError`, `ReferenceError` and `UNHANDLED` in a suite's output as failure.

```js
// Shared jsdom loader for the bug-sweep tests. Not a *.test.js file, so run-tests.js skips it.
const fs = require('fs');
const path = require('path');
const { JSDOM, VirtualConsole } = require('jsdom');

const ROOT = path.join(__dirname, '..');
const sleep = (ms) => new Promise((r) => setTimeout(r, ms));

// Boot index.html + the four app scripts, wait for app.js's own startup run to settle.
async function loadApp() {
    const html = fs
        .readFileSync(path.join(ROOT, 'index.html'), 'utf8')
        // tailwind.config is a CDN-era inline script that would throw in jsdom
        .replace(/<script>[\s\S]*?tailwind\.config[\s\S]*?<\/script>/g, '');
    const virtualConsole = new VirtualConsole();
    virtualConsole.on('jsdomError', (e) => {
        if (!String(e.message).includes('Critical Coverage Error')) console.error(e);
    });
    const dom = new JSDOM(html, {
        runScripts: 'dangerously',
        url: 'http://localhost/',
        virtualConsole,
    });
    const w = dom.window;
    w.lucide = { createIcons() {} };
    w.localStorage.setItem('schedule_lang', 'en');
    for (const f of ['structuredInputs.js', 'translations.js', 'calendarExport.js', 'app.js']) {
        const el = w.document.createElement('script');
        el.textContent = fs.readFileSync(path.join(ROOT, f), 'utf8');
        w.document.body.appendChild(el);
    }
    while (!w.initialized || w.eval('isCalculating')) await sleep(50);
    // Record toasts instead of rendering them; the real one stays on w.realShowToast
    w.toasts = [];
    w.realShowToast = w.showToast;
    w.showToast = (msg, isError) => w.toasts.push({ msg, isError: !!isError });
    return w;
}

function setInput(w, id, value) {
    const el = w.document.getElementById(id);
    if (el.type === 'checkbox') el.checked = !!value;
    else el.value = value;
}

// Known state: 4 doctors, one pool, 2 slots, Jan 2026, every optional rule off.
function reset(w, doctors = ['A', 'B', 'C', 'D']) {
    Object.entries({
        inputMonth: '1',
        inputYear: '2026',
        inputDefaultSlots: '2',
        inputOffDutyPeriod: '2',
        inputSpecialHols: '',
        inputNoDuty: '',
        inputSpecialDocs: '',
        inputSpecialDays: '0',
        inputConflicts: '',
        inputDoctorRoles: '',
        inputDefaultRoleSlots: '',
        inputRoleQuotas: '',
        inputSinglePoolQuota: '',
        inputMorningSlots: '1',
        inputAfternoonSlots: '1',
        inputNightSlots: '1',
        chkCustomDateRange: false,
        chkRoleBased: false,
        chkShiftBased: false,
        chkAllowBlankDays: false,
        chkUseSpecialRule: false,
    }).forEach(([id, v]) => setInput(w, id, v));
    w.eval(`
        doctors.length = 0; doctors.push(...${JSON.stringify(doctors)});
        offData = []; extraSlotsData = []; manualOverrides = {}; undoStack.length = 0;
        isCustomDateRange = false; globalResult = null; viewMode = 'table'; currentLang = 'en';
        scheduleDates = Array.from({ length: 31 }, (_, i) => new Date(2026, 0, i + 1));
    `);
    w.toasts.length = 0;
}

async function settle(w) {
    while (w.eval('isCalculating')) await sleep(20);
}

// JSON round trip: deepStrictEqual across the jsdom and Node realms fails on prototypes
const snap = (w, expr) => JSON.parse(w.eval(`JSON.stringify(${expr})`));

async function importJSON(w, objOrText) {
    const text = typeof objOrText === 'string' ? objOrText : JSON.stringify(objOrText);
    w.importConfigJSON({ target: { files: [new w.File([text], 'config.json')] } });
    await sleep(100); // FileReader is async
    await settle(w);
}

function runner(label) {
    let total = 0;
    let passed = 0;
    return {
        async test(name, fn) {
            total++;
            try {
                await fn();
                console.log(`✅ [PASSED] ${name}`);
                passed++;
            } catch (err) {
                console.error(`❌ [FAILED] ${name}`);
                console.error(err.message);
            }
        },
        done() {
            console.log(`\n${label}: PASSED: ${passed}/${total}, FAILED: ${total - passed}`);
            process.exit(passed === total ? 0 : 1);
        },
    };
}

module.exports = { loadApp, reset, settle, setInput, snap, importJSON, runner, sleep };
```

```js
// Validate and coerce an imported config. Pure: it never touches app state, so a bad file
// can be rejected before anything is assigned.
function sanitizeConfig(raw) {
    const isObj = (v) => v !== null && typeof v === 'object' && !Array.isArray(v);
    if (!isObj(raw)) return { ok: false, error: 'config must be a JSON object' };
    for (const key of ['doctors', 'offData', 'extraSlotsData']) {
        if (raw[key] != null && !Array.isArray(raw[key]))
            return { ok: false, error: `${key} must be an array` };
    }
    for (const key of ['manualOverrides', 'inputs', 'checkboxes']) {
        if (raw[key] != null && !isObj(raw[key]))
            return { ok: false, error: `${key} must be an object` };
    }

    const scalar = (v) => (typeof v === 'string' || typeof v === 'number' ? v : '');
    // Keep integer ids, renumber missing or repeated ones
    const withIds = (rows) => {
        const clean = rows.filter(isObj);
        let next = 1 + Math.max(0, ...clean.map((r) => (Number.isInteger(r.id) ? r.id : 0)));
        const used = new Set();
        return clean.map((r) => {
            const id = Number.isInteger(r.id) && !used.has(r.id) ? r.id : next++;
            used.add(id);
            return { ...r, id };
        });
    };

    const config = { ...raw };
    if (raw.doctors != null) {
        config.doctors = [];
        raw.doctors.forEach((d) => {
            if (typeof d !== 'string' && typeof d !== 'number') return;
            const name = sanitizeName(d);
            if (name && !config.doctors.includes(name)) config.doctors.push(name);
        });
    }
    if (raw.offData != null) {
        config.offData = withIds(raw.offData).map((r) => ({
            id: r.id,
            date: scalar(r.date),
            names: typeof r.names === 'string' ? r.names : '',
        }));
    }
    if (raw.extraSlotsData != null) {
        config.extraSlotsData = withIds(raw.extraSlotsData).map((r) => ({
            id: r.id,
            date: scalar(r.date),
            slots: scalar(r.slots),
        }));
    }
    if (raw.manualOverrides != null) {
        config.manualOverrides = {};
        Object.entries(raw.manualOverrides).forEach(([day, slots]) => {
            if (!/^\d+$/.test(day) || !isObj(slots)) return;
            const kept = {};
            Object.entries(slots).forEach(([slot, name]) => {
                if (/^\d+$/.test(slot) && typeof name === 'string') kept[slot] = name;
            });
            if (Object.keys(kept).length > 0) config.manualOverrides[day] = kept;
        });
    }
    return { ok: true, config };
}
```

```js
const CACHE_NAME = 'doctor-scheduler-v4';
const ASSETS_TO_CACHE = [
    './',
    './index.html',
    './app.js',
    './structuredInputs.js',
    './translations.js',
    './calendarExport.js',
    './dist/output.css',
    './manifest.json',
    'https://cdnjs.cloudflare.com/ajax/libs/xlsx/0.18.5/xlsx.full.min.js',
    'https://unpkg.com/lucide@1.55.0',
    'https://fonts.googleapis.com/css2?family=Sarabun:wght@300;400;500;600;700&display=swap',
];

self.addEventListener('install', (event) => {
    self.skipWaiting();
    event.waitUntil(
        caches.open(CACHE_NAME).then((cache) =>
            // One failed CDN request must not abort the whole install
            Promise.allSettled(ASSETS_TO_CACHE.map((url) => cache.add(url)))
        )
    );
});

self.addEventListener('activate', (event) => {
    event.waitUntil(
        caches
            .keys()
            .then((names) =>
                Promise.all(names.filter((n) => n !== CACHE_NAME).map((n) => caches.delete(n)))
            )
            .then(() => self.clients.claim())
    );
});

self.addEventListener('fetch', (event) => {
    const req = event.request;
    if (req.method !== 'GET') return;
    const sameOrigin = new URL(req.url).origin === self.location.origin;
    const cached = (r) => caches.match(r).then((hit) => hit || Promise.reject(new Error('miss')));
    const network = () =>
        fetch(req).then((res) => {
            if (res.ok) {
                const copy = res.clone();
                caches.open(CACHE_NAME).then((c) => c.put(req, copy));
            }
            return res;
        });
    // Our own files: network first, so a deploy shows on the next load and offline still works.
    // CDN files are version-pinned, so the cache is always right.
    const response = sameOrigin ? network().catch(() => cached(req)) : cached(req).catch(network);
    event.respondWith(
        response.catch(() =>
            req.mode === 'navigate' ? caches.match('./index.html') : Response.error()
        )
    );
});
```
